// === EDITOR API (gọi qua window.editor) ===
editor.clear()                              // xóa toàn bộ canvas (cells = empty)
editor.getSize()                            // trả {w, h}
editor.setSize(w, h)                        // đổi kích cỡ (giữ data nếu có thể)
editor.setCell(x, y, charOrCp, fg, bg)      // đặt 1 ô
  // charOrCp: string 1 ký tự ('#') HOẶC int codepoint (0x2588)
  // fg:       0-15 (CGA). Bỏ qua = 15 (trắng)
  // bg:       0-15, -1, hoặc bỏ qua = transparent
editor.setGrid(rows)                        // đặt cả grid. rows = [[cell,...],...] row-major
  // mỗi cell: {} = transparent, hoặc {c:'█',fg:7,bg:0}
  // c: char string hoặc codepoint int. fg mặc định 15, bg mặc định transparent
editor.fillRect(x0,y0,x1,y1, char,fg,bg)    // fill hình chữ nhật
editor.fillGlyph(charOrCp, fg, bg)          // fill toàn bộ canvas cùng 1 cell
editor.getGrid()                            // trả rows array (deep copy)
editor.getJS()                              // trả string JS code tái tạo grid hiện tại
editor.getC([name])                         // trả string C array (mặc định name="SPRITE")
editor.exportXp([filename])                 // trigger download file .xp
editor.loadXp(arrayBuffer)                  // load .xp từ ArrayBuffer (async)

// === VÍ DỤ 1: bức tường đá 3x3 ===
editor.setSize(3,3);
editor.setGrid([
  [{c:'▓',fg:8}, {c:'▓',fg:7}, {c:'░',fg:8}],
  [{c:'▓',fg:7}, {c:'▒',fg:7}, {c:'▓',fg:8}],
  [{c:'░',fg:8}, {c:'▓',fg:8}, {c:'▓',fg:7}]
]);

// === VÍ DỤ 2: potion 2x2 ===
editor.setSize(2,2);
editor.setGrid([
  [{c:'╔',fg:7}, {c:'╗',fg:7}],
  [{c:'║',fg:4,bg:4}, {c:'║',fg:4,bg:4}]
]);

// === VÍ DỤ 3: dùng codepoint trực tiếp ===
editor.setCell(0, 0, 0x2588, 12, 0);   // █ đỏ trên nền đen
Sẵn sàng. 3×3 grid. Click palette → chọn màu. Click glyph → chọn ký tự. Click canvas = vẽ.
